<script>
let clickCount = 0;

function catMagicClick() {
  clickCount++; // クリックされたらすぐに数を数える

  // ★ 5回に1回、肉球スタンプを出す！
  if (clickCount % 5 === 0) {
    showPawStamp();
  }

  // ★ 4パターンループに変更しました！
  const pattern = (clickCount - 1) % 4;
  if (pattern === 0) {
    releaseCats('circular-spin'); // ★追加：画面中央で輪になって回転（5秒）
  } else if (pattern === 1) {
    releaseCats('bl-to-tr'); // 左下→右上
  } else if (pattern === 2) {
    releaseCats('tr-to-bl'); // 右上→左下
  } else if (pattern === 3) {
    releaseCats('bottom-to-top'); // 下→上
  }
}

function showPawStamp() {
  const container = document.getElementById('cat-horde-container');
  const stamp = document.createElement('div');
  stamp.className = 'paw-stamp-overlay';
  container.appendChild(stamp);
  setTimeout(() => {
    stamp.remove();
  }, 2000);
}

function releaseCats(direction) {
  const container = document.getElementById('cat-horde-container');
  const catUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/ezgif.com-animated-gif-maker.gif'; 
  
  // ★ ▼ ここに一匹だけ混ぜる別の生き物のURLを貼り付け ▼ ★
  const otherImageUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/ezgif.com-animated-gif-maker.gif'; 

  // ★ 負荷を下げるため「circular-spin」パターンでは輪を構成する「16匹」に、
  // 　 それ以外はこれまでの「10匹」にします
  let catCount, duration;
  
  if (direction === 'circular-spin') {
    catCount = 16;
    duration = 5; // ★ 5秒固定
  } else {
    catCount = 10;
    duration = null; // ランダム
  }
  
  const otherIndex = Math.floor(Math.random() * catCount);

  // ★ 輪になって回るための親コンテナを作成（画面の中心に配置）
  let spinContainer = null;
  if (direction === 'circular-spin') {
    spinContainer = document.createElement('div');
    spinContainer.style.position = 'absolute';
    spinContainer.style.top = '50%';
    spinContainer.style.left = '50%';
    spinContainer.style.transform = 'translate(-50%, -50%) rotate(0deg)';
    spinContainer.style.width = '1px'; // 中心点として機能
    spinContainer.style.height = '1px';
    // ★コンテナも GPUで軽く動かす
    spinContainer.style.willChange = 'transform';
    container.appendChild(spinContainer);
  }

  for (let i = 0; i < catCount; i++) {
    // 出現タイミングを少し早くしてサッと現れさせる
    let delay = direction === 'circular-spin' ? i * (1000 / catCount) : Math.random() * 800;
    
    setTimeout(() => {
      const cat = document.createElement('div');
      cat.className = 'horde-cat';
      
      // 画像の振り分け
      if (i === otherIndex) {
        cat.style.backgroundImage = `url('${otherImageUrl}')`;
      } else {
        cat.style.backgroundImage = `url('${catUrl}')`;
      }

      // サイズ設定（40px〜120px、輪移動では少しメリハリを出す）
      const size = Math.floor(Math.random() * 80) + 40;
      cat.style.width = `${size}px`;
      cat.style.height = `${size}px`;

      // 速度設定
      const finalDuration = duration ? duration : (Math.random() * 1.8) + 1.2;
      
      let startX, startY, endX, endY, flip;

      // ★ 輪になって回るパターンの配置計算
      if (direction === 'circular-spin') {
        const R = 30; // 半径（画面幅の30%、vminで自動調整）
        const theta = i * (360 / catCount); // 角度（度）
        
        // ★ JSで配置：中心からR離れた円周上に、進行方向を向くように配置
        cat.style.left = '50%';
        cat.style.top = '50%';
        // 初期位置と向きをセット
        cat.style.transformOrigin = 'center center';
        // 時計回り、進行方向向き： rotate(${theta}deg) で角度 thetaへ配置、さらに rotate(90deg)で進行方向を向く
        cat.style.transform = `translate(-50%, -50%) rotate(${-theta}deg) translate(${R}vmin) rotate(90deg) scaleX(1)`;
        
        // アニメーションは親コンテナに設定するので、猫自身のアニメーションは不要
        spinContainer.appendChild(cat);
        
      } else {
        // 直線移動パターンの計算
        if (direction === 'bl-to-tr') {
          startX = Math.random() * 50 - 30; startY = Math.random() * 50 + 80; endX = Math.random() * 50 + 80; endY = Math.random() * 50 - 30; flip = 1; 
        } else if (direction === 'tr-to-bl') {
          startX = Math.random() * 50 + 80; startY = Math.random() * 50 - 30; endX = Math.random() * 50 - 30; endY = Math.random() * 50 + 80; flip = -1; 
        } else if (direction === 'bottom-to-top') {
          startX = Math.random() * 120 - 10; startY = 120; endX = startX + (Math.random() * 40 - 20); endY = -30; flip = Math.random() > 0.5 ? 1 : -1; 
        }
        
        container.appendChild(cat);
        // ★ブラウザの負担を最小限にするGPUアニメーション（直線移動）
        cat.animate([
          { transform: `translate3d(${startX}vw, ${startY}vh, 0) scaleX(${flip})` },
          { transform: `translate3d(${endX}vw, ${endY}vh, 0) scaleX(${flip})` }
        ], {
          duration: finalDuration * 1000,
          easing: 'linear',
          fill: 'forwards'
        });
        
        // 直線移動した猫は走り終わったら削除
        setTimeout(() => {
          cat.remove();
        }, finalDuration * 1000);
      }

    }, delay); 
  }

  // ★輪移動パターンのアニメーション：親コンテナを 5秒で1回転させる
  if (direction === 'circular-spin') {
    // 画面中央を中心に 360度回転
    spinContainer.animate([
      { transform: 'translate(-50%, -50%) rotate(0deg)' },
      { transform: 'translate(-50%, -50%) rotate(360deg)' } // 1回転
    ], {
      duration: duration * 1000,
      easing: 'linear',
      fill: 'forwards'
    });
    
    // 5秒（アニメーション終了後）に親コンテナごと猫たちを削除
    setTimeout(() => {
      spinContainer.remove();
    }, duration * 1000);
  }
}
</script>

<p>例えば骨材全部の質量(m<sub>all</sub>)が100ｇで、最も目の大きな40ｍｍのふるいに留まった骨材の質量(m<sub>40</sub>)が20ｇ、次に大きい20ｍｍのふるいに溜まった骨材の質量(m<sub>20</sub>)が10ｇの場合、以下のような計算になります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>X40=m40/mall=20/100=20(％)</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>X20=（m40+m20）/mall=（20+10）/100=20+10=30(％)<br>ということになります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>ふるいに留まった骨材の質量百分率の上記の表に当てはまれば、それはふるい分けにおいては、JIS相当品になります。</p>
