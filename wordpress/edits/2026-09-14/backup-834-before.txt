function catMagicClick() {
  // もし大群が走っている最中なら、クリックを無視する（重くなるのを防ぐ）

  const pattern = clickCount % 3;

  } else if (pattern === 1) {

  } else if (pattern === 2) {

  clickCount++;

function releaseCats(direction) {
  isRunning = true; // 大群発生スタート！追加クリックを禁止する

  const container = document.getElementById('cat-horde-container');
  // ▼ ここに通常の大群用の猫のURLを貼り付け ▼

  const otherIndex = Math.floor(Math.random() * catCount);

    setTimeout(() => {

        cat.style.backgroundImage = `url('${catUrl}')`;
      }

      const size = Math.floor(Math.random() * 70) + 50; // サイズもランダム
      cat.style.width = `${size}px`;
      cat.style.height = `${size}px`;

      const duration = (Math.random() * 2.5) + 1.5; // 速度もランダム

      if (direction === 'left-to-right') {
        cat.style.top = `${Math.floor(Math.random() * 90)}vh`;
        cat.style.left = '-200px';

<p>この資料は、トラックドライバーの労働環境改善を目指し、国土交通省が荷主に対して行う是正指導の運用指針をまとめたものです。長時間の荷待ちや不当な運賃の据え置きなど、物流の停滞を招く違反原因行為を定義し、それらに対する「働きかけ」「要請」「勧告」という三段階の指導プロセスを詳しく規定しています。専門組織であるトラック・物流Gメンによる調査や、情報の提供者保護、さらには悪質なケースにおける荷主名の公表についても具体的な基準が示されています。また、行政指導の透明性を確保するための様式例や、関係省庁との連携体制も網羅されています。全体として、荷主と運送事業者の適正な取引環境を構築し、持続可能な物流を実現するための実務的なガイドラインとなっています。</p>

<!-- wp:heading -->
<h2 class="wp-block-heading">Q. トラック・物流Gメン による是正指導の仕組みを教えて</h2>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>A.トラック・物流Gメンによる荷主への是正指導は、トラックドライバーの労働条件改善と物流の適正化を目的としており、主に<strong>「働きかけ」「要請」「勧告」の3段階の措置</strong>で構成されています 。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>以下にその具体的な仕組みを解説します。</p>
<!-- /wp:paragraph -->

<!-- wp:heading {"fontSize":"medium"} -->
<h2 class="wp-block-heading has-medium-font-size">1. トラック・物流Gメンによる是正指導：3つのステップ</h2>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>荷主による「違反原因行為（運送事業者が法令違反をする原因となる行為）」の疑いの程度や状況に応じて、以下の指導が行われます。</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li><strong>働きかけ:</strong> 荷主が違反原因行為をしている<strong>疑いがある場合</strong>に実施されます 。荷主に対し、運送事業者が法令を遵守できるよう配慮することの重要性を伝え、<strong>自主的な確認と改善</strong>を促す文書を発出します 。この段階では、行政側から改善報告を求めることはありません 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>要請:</strong> 違反原因行為をしていると疑うに足りる<strong>相当な理由がある場合</strong>に行われる、より強い措置です 。荷主に対して違反原因行為をしないよう正式に求め、<strong>改善計画の作成と報告</strong>を義務付けます 。大企業や悪質なケースではヒアリングも実施されます 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>勧告:</strong> 要請を受けた後も<strong>なお違反原因行為を継続している</strong>と認められる場合、あるいは改善計画の提出を拒む場合などに行われます 。勧告がなされると、<strong>荷主名と内容が公表される</strong>ため、社会的な影響が非常に大きい措置です 。</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:heading {"level":3,"fontSize":"medium"} -->
<h3 class="wp-block-heading has-medium-font-size">2. 指導の端緒となる情報収集（Gメン調査員・意見募集）</h3>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>トラック・物流Gメンは、以下のルートで得た情報を元に調査を開始します。</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li><strong>意見募集窓口:</strong> 国土交通省のホームページに設置された窓口への投稿 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>実態調査:</strong> トラック業者や倉庫業者へのヒアリング、荷主へのパトロール 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>Gメン調査員:</strong> 各都道府県のトラック協会に配置された「Gメン調査員」からの情報提供 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>街頭活動:</strong> サービスエリアや物流拠点付近でのトラックドライバーへの聞き取り 。</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:heading {"level":3,"fontSize":"medium"} -->
<h3 class="wp-block-heading has-medium-font-size">3. 荷主による「違反原因行為」の具体例（荷待ち・不当な据置きなど）</h3>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>是正指導の対象となる主な行為は以下の通りです。</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li><strong>長時間の荷待ち:</strong> 恒常的に<strong>1時間以上</strong>（荷役込みで2時間以上）待機させている場合 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>契約にない附帯業務:</strong> 仕分け、棚入れ、ラベル貼りなどを、契約や対価の合意なしに行わせること 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>運賃・料金の不当な据置き:</strong> 燃料費高騰などの根拠を示した交渉に応じず、一方的に運賃を据え置くこと 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>過積載運送の指示・容認:</strong> 最大積載量を超える荷積みを指示、または知りながら容認すること 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>異常気象時の運送依頼:</strong> 台風や大雪などで安全確保が困難な状況下で、運送を強要すること 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>無理な運送依頼:</strong> 法定の拘束時間を超えなければ不可能な配送指示や、高速代を払わず高速利用を強いる時間設定など 。</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph -->
<p>なお、運賃が据え置かれた結果として適正原価を下回り続けた場合、荷主に対しては働きかけや勧告にとどまる一方で、<a href="https://www.ishinazaka.co.jp/tekisei-genka-kyoka-koushin/">許可を更新できなくなるのは運送事業者の側</a>という仕組みが検討されています。</p>
<!-- /wp:paragraph -->

<!-- wp:heading {"fontSize":"medium"} -->
<h2 class="wp-block-heading has-medium-font-size">4. 特徴的な仕組み</h2>
<!-- /wp:heading -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li><strong>体制の強化:</strong> 2024年11月に「トラックGメン」から「トラック・物流Gメン」に改組され、倉庫業者からの情報収集も対象に加わるとともに、<strong>総勢360名規模</strong>へ体制が拡充されました 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>情報提供者の保護:</strong> 指導の際、情報提供者が特定されないよう十分留意されます [23]。また、提供者が希望すれば、どのような指導を行ったかの結果通知を受けることも可能です 。</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li><strong>関係機関との連携:</strong> 国土交通省だけでなく、荷主を所管する経済産業省や農林水産省、労働基準法を所管する厚生労働省、不公正な取引を監視する公正取引委員会などと情報を共有し、連携して対応します 。</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph -->
<p>個別の事案について貨物自動車運送事業法上（以下「法」という。）の許可の要否について判断に迷われる場合には、各運輸局にご相談ください。</p>
<!-- /wp:paragraph -->
