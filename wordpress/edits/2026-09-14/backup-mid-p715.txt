<script>
let clickCount = 0;
// ★ クリックを邪魔していた isRunning を完全に削除しました！いつでもクリックできます。

function catMagicClick() {
  clickCount++; // クリックされたらすぐに数を数える

  // ★ 5回に1回、肉球スタンプを出す！
  if (clickCount % 5 === 0) {
    showPawStamp();
  }

  // 3パターンループ
  const pattern = (clickCount - 1) % 3;
  if (pattern === 0) {
    releaseCats('bl-to-tr'); // 1回目：左下→右上
  } else if (pattern === 1) {
    releaseCats('tr-to-bl'); // 2回目：右上→左下
  } else if (pattern === 2) {
    releaseCats('bottom-to-top'); // 3回目：下→上
  }
}

function showPawStamp() {
  const container = document.getElementById('cat-horde-container');
  const stamp = document.createElement('div');
  stamp.className = 'paw-stamp-overlay';
  
  container.appendChild(stamp);

  // スタンプが消えたらデータごと削除して軽くする
  setTimeout(() => {
    stamp.remove();
  }, 2000);
}

function releaseCats(direction) {
  const container = document.getElementById('cat-horde-container');
  const catUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/ezgif.com-animated-gif-maker.gif'; 
  
  // ★ ▼ ここに一匹だけ混ぜる別の生き物のURLを貼り付け ▼ ★
  const otherImageUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/ezgif.com-animated-gif-maker.gif'; 

  const catCount = 10; 
  const otherIndex = Math.floor(Math.random() * catCount);

  for (let i = 0; i < catCount; i++) {
    setTimeout(() => {
      const cat = document.createElement('div');
      cat.className = 'horde-cat';
      
      // 画像の振り分け
      if (i === otherIndex) {
        cat.style.backgroundImage = `url('${otherImageUrl}')`;
      } else {
        cat.style.backgroundImage = `url('${catUrl}')`;
      }

      const size = Math.floor(Math.random() * 80) + 40;
      cat.style.width = `${size}px`;
      cat.style.height = `${size}px`;

      const duration = (Math.random() * 1.8) + 1.2;
      let startX, startY, endX, endY, flip;

      if (direction === 'bl-to-tr') {
        startX = Math.random() * 50 - 30; startY = Math.random() * 50 + 80; endX = Math.random() * 50 + 80; endY = Math.random() * 50 - 30; flip = 1; 
      } else if (direction === 'tr-to-bl') {
        startX = Math.random() * 50 + 80; startY = Math.random() * 50 - 30; endX = Math.random() * 50 - 30; endY = Math.random() * 50 + 80; flip = -1; 
      } else if (direction === 'bottom-to-top') {
        startX = Math.random() * 120 - 10; startY = 120; endX = startX + (Math.random() * 40 - 20); endY = -30; flip = Math.random() > 0.5 ? 1 : -1; 
      }

      container.appendChild(cat);

      cat.animate([
        { transform: `translate3d(${startX}vw, ${startY}vh, 0) scaleX(${flip})` },
        { transform: `translate3d(${endX}vw, ${endY}vh, 0) scaleX(${flip})` }
      ], {
        duration: duration * 1000,
        easing: 'linear',
        fill: 'forwards'
      });

      setTimeout(() => {
        cat.remove();
      }, duration * 1000);

    }, Math.random() * 800); // すぐに出現するように調整
  }
}

<h2 class="wp-block-heading">自家用ダンプカーの貨物自動車運送事業法における取扱いについて（国土交通省）</h2>
<!-- /wp:heading -->

<figure class="wp-block-image size-large is-resized"><img src="https://www.ishinazaka.co.jp/common/files/uploads/2026/03/1000006594-1024x772.jpg" alt="自家用ダンプカーの取扱いに関する資料" class="wp-image-730" style="width:400px"/></figure>

<!-- wp:paragraph -->
<p>令和8年4月1日から、いわゆる「白トラ」に運送委託を行った荷主等に対する規制が適用されることに伴い、自家用ダンプカーの貨物自動車運送事業法における取扱いが下記のとおり明確化されました。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"align":"center"} -->
<p class="has-text-align-center">記</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>　建設現場等で使用するダンプカーについては、①他人の需要に応じ、②有償で、③貨物の運送を事業として行う場合には、貨物自動車運送事業法（以下 法」という。）の許可が必要となりますが、下記の１．（１）又は（２）のいずれかに該当し、２．の要件を具備した場合には、法の許可が不要となります。なお、個別の事案の判断に当たっては、下記を参照いただいた上で、判断に迷われる場合には、地方運輸局までご相談ください。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>　また、法の許可が不要となる場合であっても、運転業務に主として従事する労働者については、「自動車運転者の労働時間等の改善のための基準」（改善基準告示）が適用されることにご留意ください。</p>
<!-- /wp:paragraph -->

<!-- wp:heading {"level":3} -->
<h3 class="wp-block-heading"><strong>１．法の許可が不要となる運送</strong></h3>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>（１）建設関連会社等が自ら所有する貨物を自ら運送する場合</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>　自ら所有する貨物を自ら運送する場合には、自社のニーズや発意に応じて運送が行われることが通常であり、運送行為の対価も発生しないことから、上記①及び②に該当せず、法の許可は不要となる。<br>＜具体例＞<br>・土砂等販売業者が、販売するために購入した土砂等を、自社と雇用関係にある従業員（期間雇用又は日雇い雇用等の場合を含む。）に運搬させる場合</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>（２）建設関連会社等の生業と密接不可分であり、その業務に付帯するものとして運送を行う場合</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>　他者が所有する貨物であっても、下記ⅰ）～ⅲ）のいずれにも該当する場合には、業としての運送を行っているとは言えず、上記②及び③に該当しないと整理できることから、法の許可は不要となる。<br>ⅰ）建設関連会社等の生業と密接不可分であり、その業務に付帯して行われる運送であること<br>ⅱ）上記ⅰ）の生業に付帯して行われる運送と認められるための具備要件として、当該生業を営む建設関連会社等が自ら運送行為を行うこと（同一の者が当該生業と当該運送行為とを一貫して行うこと）<br>ⅲ）名目の如何を問わず、運送行為の対価としての有償性がないこと<br>＜具体例＞<br>・建設工事を請け負った建設関連会社等が、自社の行う建設工事に付帯する業務として、当該建設工事で発生する残土等を、自社と雇用関係にある従業員（期間雇用又は日雇い雇用等の場合を含む。）に運搬させる場合<br>・土砂等販売を代行する個人事業主が、当該個人事業主の行う土砂等販売代行に付帯する業務として、販売する土砂等を当該個人事業主が運搬する場合</p>
<!-- /wp:paragraph -->

<!-- wp:heading {"level":3} -->
<h3 class="wp-block-heading"><strong>２．自ら運送を行っていると認められるための具備要件</strong></h3>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>　上記１．（１）（２）において、建設関連会社等が自ら運送を行っていると認められるためには、当該会社等と雇用関係にある従業員たる運転者（期間雇用又は日雇い雇用等の場合を含む。）に運送行為を行わせることが必要である。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>　雇用関係があるか否かについては、契約等の形態のみならず、使用従属性等の実態も踏まえて判断されることとなる。少なくとも主な判断基準としては、<br>・建設関連会社等と運転者との間で労働契約が締結されているか<br>・運転者に対して労働条件通知書の交付がなされているか<br>・運転者に対する報酬が給与として支払われているか<br>・社会保険等の加入が必要な場合に社会保険等の加入や支払い等の適切な措置が講じられているか<br>・運転者が持ち込む自家用ダンプカーを使用する場合、運転者と建設関連会社等との間で、当該車両の業務上使用契約書の締結等の適切な措置が講じられているか<br>・運転者が当該建設関連会社等の指揮命令下にあるか 等<br>があるが、労働契約や労働条件通知書等に関する詳細は、最寄りの労働局・労働基準監督署にご確認いただきたい。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>すべての白ダンプが違法ではない。上記の例に当てはまるのかどうかは、国土交通省等へ問い合わせてください。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>国は様々な具体的なケースを示し、違法か合法を示さないと、白トラの不要な廃業を招く可能性がある。分かりやすい情報提供をお願いしたい。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>ここまでは「許可が要るか、要らないか」の話でした。許可を取ったあとについては、事業許可を5年ごとの更新制にする改正が別に進んでいます（改正法は公布済み、2026年9月時点で施行日は未定）。<a href="https://www.ishinazaka.co.jp/kensetsugyo-unso-kyoka-koushin/">許可を取ったあと、 運送許可 の更新で何が問われるのか</a>を、白ナンバー側にやることが残るのかも含めて書きました。</p>
<!-- /wp:paragraph -->
