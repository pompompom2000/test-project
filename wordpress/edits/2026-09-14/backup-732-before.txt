<script>
let clickCount = 0;
let isRunning = false; // 大群が走っている最中かどうかを判定するフラグ

function catMagicClick() {
  // もし大群が走っている最中なら、クリックを無視する（重くなるのを防ぐ）
  if (isRunning) return; 

  const pattern = clickCount % 3;
  
  if (pattern === 0) {
    releaseCats('left-to-right');
  } else if (pattern === 1) {
    releaseCats('right-to-left');
  } else if (pattern === 2) {
    releaseCats('top-to-bottom');
  }
  
  clickCount++;
}

function releaseCats(direction) {
  isRunning = true; // 大群発生スタート！追加クリックを禁止する

  const container = document.getElementById('cat-horde-container');
  // ▼ ここに通常の大群用の猫のURLを貼り付け ▼
  const catUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/ezgif.com-animated-gif-maker.gif'; 
  
  // ★ ▼ ここに【一匹だけ混ぜる別の生き物】のGIF画像のURLを貼り付け ▼ ★
  const otherImageUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/1775735437402.png'; 

  const catCount = 20; // 猫の数

  // 大群（catCount匹）の中から、ランダムに「一匹」を選ぶ
  const otherIndex = Math.floor(Math.random() * catCount);

  for (let i = 0; i < catCount; i++) {
    setTimeout(() => {
      const cat = document.createElement('div');
      cat.className = 'horde-cat';

      // 通常は猫の画像、選ばれた一匹だけ別の生き物の画像にする
      if (i === otherIndex) {
        cat.style.backgroundImage = `url('${otherImageUrl}')`;
      } else {
        cat.style.backgroundImage = `url('${catUrl}')`;
      }

      const size = Math.floor(Math.random() * 70) + 50; // サイズもランダム
      cat.style.width = `${size}px`;
      cat.style.height = `${size}px`;

      const duration = (Math.random() * 2.5) + 1.5; // 速度もランダム

      if (direction === 'left-to-right') {
        cat.style.top = `${Math.floor(Math.random() * 90)}vh`;
        cat.style.left = '-200px';
        cat.style.transform = 'scaleX(1)'; // 右向き
        cat.style.animation = `horde-run-lr ${duration}s linear forwards`;
      } else if (direction === 'right-to-left') {
        cat.style.top = `${Math.floor(Math.random() * 90)}vh`;
        cat.style.right = '-200px';
        cat.style.transform = 'scaleX(-1)'; // 左向き
        cat.style.animation = `horde-run-rl ${duration}s linear forwards`;
      } else if (direction === 'top-to-bottom') {
        cat.style.left = `${Math.floor(Math.random() * 95)}vw`;
        cat.style.top = '-200px';
        const flip = Math.random() > 0.5 ? 1 : -1; // 落ちるときは向きもランダム
        cat.style.transform = `scaleX(${flip})`;
        cat.style.animation = `horde-fall-tb ${duration}s linear forwards`;
      }

      container.appendChild(cat);

      setTimeout(() => {
        cat.remove(); // 走り終わった猫を消去
      }, duration * 1000);

    }, Math.random() * 1500); // 出現のバラつき
  }

  // 大群が全員走り終わるころ（約4.5秒後）に、クリック禁止を解除する
  setTimeout(() => {
    isRunning = false;
  }, 4500);
}
</script>

<p>この資料は、改正された物資の流通の効率化に関する法律に基づき、一定規模以上の貨物を取り扱う特定荷主が取り組むべき実務手続きを解説したガイドブックです。物流分野における労働環境の改善や輸送能力の維持を目的としており、中長期計画書の作成や定期報告書の提出、物流統括管理者の選任といった具体的な義務内容を詳述しています。特に、トラック運転手の荷待ち時間や荷役時間の短縮、および積載効率の向上に向けた判断基準が具体的に示されているのが特徴です。対象となる荷主は、前年度の取扱重量が9万トンを超える事業者であり、義務を怠った場合には罰則や勧告の対象となることが明記されています。本書は、2025年9月の施行に向け、企業が法制度を遵守しつつ自律的に物流の効率化を進めるための指針としてまとめられています。</p>
<!-- /wp:paragraph -->

<!-- wp:heading {"level":3} -->
<h3 class="wp-block-heading"><strong>Q.特定荷主に指定される基準と、必要な届出の手順を教えてください。</strong></h3>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>A.特定荷主に指定される基準と、その届出に関する手順について、提供された資料に基づき説明します。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>特定荷主に指定される基準</strong></p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>特定荷主とは、自らの事業に関して継続的に貨物の運送を行わせる(第一種荷主)、または貨物の受渡しを行う(第二種荷主)者のうち、その規模が一定以上の事業者を指します。</p>
<!-- /wp:paragraph -->

<!-- wp:image {"id":832,"width":"400px","sizeSlug":"full","linkDestination":"none"} -->
<figure class="wp-block-image size-full is-resized"><img src="https://www.ishinazaka.co.jp/common/files/uploads/2026/03/1000006619.png" alt="1000006619" class="wp-image-832" style="width:400px"/></figure>
<!-- /wp:image -->

<!-- wp:paragraph -->
<p><strong>基準重量</strong>： 前年度の取扱貨物重量の合計が、第一種荷主または第二種荷主のいずれかの区分で 9万トン以上 であることが基準となります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>算定の対象</strong></p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>。第一種荷主:貨物自動車運送事業者などに運送を行わせた貨物の合計重量。<br>第二種荷主:自らの事業に関して、運転者(他者に雇用されている者に限る)から受け取る、または運転者に引き渡す貨物の合計重量。ただし、自ら運送を委託する場合(第一種荷主に該当)や、受渡しの時間帯を指示できない場合は除かれ ます。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>除外できるもの</strong>： 郵便物、信書便物、30kg以下の特別宅配貨物（一定条件あり）、および取扱貨物合計の1％程度の軽量な資材や事務用品などは算定から除くことができます。</p>
<!-- /wp:paragraph -->

<!-- wp:heading {"level":3} -->
<h3 class="wp-block-heading"><strong>必要な届出の手順</strong></h3>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>取扱貨物重量が基準（9万トン）を超えた場合、以下の手順で手続きを行う必要があります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>1.貨物の運送の委託及び受渡しの状況届出 (様式第1)</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>初めて基準重量を上回った年度の翌年度の5月末日までに、「貨物の運送の委託及び受渡しの状況届出書(様式第1) 」を提出します。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><br>·提出先:貨物の運送委託や受渡しを伴う事業を所管する 全ての大臣です。<br>·提出方法: 原則として、届出システムによるオンライン提出となりますが、オンラインが困難な場合は郵送も可能です。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>2. 指定後の手続き</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>特定荷主として指定された後は、以下の対応が義務付けられます。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>・物流統括管理者の選任と届出(様式第4) :指定通知を受けた後、すみやかに事業運営上の重要な決定に参画する管理的地位にある者を選任し、届け出る必要があり ます。<br></p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>・中長期計画書の作成·提出(様式第3)<br>運転者の荷待ち時間短縮や積載効率向上などの取組に関する計画を作成し、提出します。初回期限は指定年度の7月末(2026年 度は10月末)です。 </p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>・定期報告書の提出(様式第5) :指定を受けた年度の翌年度から、毎年度7月末日までに前年度の取組状況などを報告します。 </p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>なお、これらの届出や報告を行わない、あるいは虚偽の届出をした場合には、50万円以下の罰金 などの罰則が科される可能性があります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>詳細なガイドラインは、<a href="https://www.revised-logistics-act-portal.mlit.go.jp/sippers/" target="_blank" rel="noreferrer noopener">物流効率化法理解促進ポータルサイト</a>や、管轄の経済産業局の情報を必ず確認してください。&nbsp;</p>
