<!-- wp:heading -->
<h2 class="wp-block-heading">荷主の区分の考え方</h2>
<!-- /wp:heading -->

<!-- wp:paragraph -->
<p>貨物自動車運送事業法における「荷主」には、大きく分けて第1種荷主と第2種荷主の2つの区分があります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>第1種荷主とは：</strong> 運送事業者と直接運送契約を結ぶ者のことを指します。自らの荷物を運んでもらう依頼主がこれに該当します。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p><strong>第2種荷主とは：</strong> 第1種荷主から運送の依頼を受け、さらに別の運送事業者へ運送を依頼する者（いわゆる利用運送事業者など）を指します。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph -->
<p>建設現場や砕石の運搬においても、誰が契約の主体となるかによってこの区分が重要になる場合があります。法令を遵守し、適切な運行管理を行うためにも、自社がどの区分に該当するかを正しく把握しておくことが大切です。判断に迷う場合は、管轄の運輸局へ相談することをお勧めします。</p>
<!-- /wp:paragraph -->

<script>
let clickCount = 0;
let isRunning = false; // 大群が走っている最中かどうかを判定するフラグ

function catMagicClick() {
  // もし大群が走っている最中なら、クリックを無視する（重くなるのを防ぐ）
  if (isRunning) return; 

  const pattern = clickCount % 3;
  
  if (pattern === 0) {
    releaseCats('left-to-right');
  } else if (pattern === 1) {
    releaseCats('right-to-left');
  } else if (pattern === 2) {
    releaseCats('top-to-bottom');
  }
  
  clickCount++;
}

function releaseCats(direction) {
  isRunning = true; // 大群発生スタート！追加クリックを禁止する

  const container = document.getElementById('cat-horde-container');
  // ▼ ここに通常の大群用の猫のURLを貼り付け ▼
  const catUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/ezgif.com-animated-gif-maker.gif'; 
  
  // ★ ▼ ここに【一匹だけ混ぜる別の生き物】のGIF画像のURLを貼り付け ▼ ★
  const otherImageUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/04/1775735437402.png'; 

  const catCount = 20; // 猫の数

  // 大群（catCount匹）の中から、ランダムに「一匹」を選ぶ
  const otherIndex = Math.floor(Math.random() * catCount);

  for (let i = 0; i < catCount; i++) {
    setTimeout(() => {
      const cat = document.createElement('div');
      cat.className = 'horde-cat';

      // 通常は猫の画像、選ばれた一匹だけ別の生き物の画像にする
      if (i === otherIndex) {
        cat.style.backgroundImage = `url('${otherImageUrl}')`;
      } else {
        cat.style.backgroundImage = `url('${catUrl}')`;
      }

      const size = Math.floor(Math.random() * 70) + 50; // サイズもランダム
      cat.style.width = `${size}px`;
      cat.style.height = `${size}px`;

      const duration = (Math.random() * 2.5) + 1.5; // 速度もランダム

      if (direction === 'left-to-right') {
        cat.style.top = `${Math.floor(Math.random() * 90)}vh`;
        cat.style.left = '-200px';
        cat.style.transform = 'scaleX(1)'; // 右向き
        cat.style.animation = `horde-run-lr ${duration}s linear forwards`;
      } else if (direction === 'right-to-left') {
        cat.style.top = `${Math.floor(Math.random() * 90)}vh`;
        cat.style.right = '-200px';
        cat.style.transform = 'scaleX(-1)'; // 左向き
        cat.style.animation = `horde-run-rl ${duration}s linear forwards`;
      } else if (direction === 'top-to-bottom') {
        cat.style.left = `${Math.floor(Math.random() * 95)}vw`;
        cat.style.top = '-200px';
        const flip = Math.random() > 0.5 ? 1 : -1; // 落ちるときは向きもランダム
        cat.style.transform = `scaleX(${flip})`;
        cat.style.animation = `horde-fall-tb ${duration}s linear forwards`;
      }

      container.appendChild(cat);

      setTimeout(() => {
        cat.remove(); // 走り終わった猫を消去
      }, duration * 1000);

    }, Math.random() * 1500); // 出現のバラつき
  }

  // 大群が全員走り終わるころ（約4.5秒後）に、クリック禁止を解除する
  setTimeout(() => {
    isRunning = false;
  }, 4500);
}
</script>
