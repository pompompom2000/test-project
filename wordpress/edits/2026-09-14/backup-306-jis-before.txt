<script>
let clickCount = 0;
// ★ クリックを邪魔していた isRunning を完全に削除しました！いつでもクリックできます。

function catMagicClick() {
  clickCount++; // クリックされたらすぐに数を数える

  // ★ 5回に1回、肉球スタンプを出す！
  if (clickCount % 5 === 0) {
    showPawStamp();
  }

  // 3パターンループ
  const pattern = (clickCount - 1) % 3;
  if (pattern === 0) {
    releaseCats('bl-to-tr'); // 1回目：左下→右上
  } else if (pattern === 1) {
    releaseCats('tr-to-bl'); // 2回目：右上→左下
  } else if (pattern === 2) {
    releaseCats('bottom-to-top'); // 3回目：下→上
  }
}

function showPawStamp() {
  const container = document.getElementById('cat-horde-container');
  const stamp = document.createElement('div');
  stamp.className = 'paw-stamp-overlay';
  
  container.appendChild(stamp);

  // スタンプが消えたらデータごと削除して軽くする
  setTimeout(() => {
    stamp.remove();
  }, 2000);
}

function releaseCats(direction) {
  const container = document.getElementById('cat-horde-container');
  const catUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/05/1000006878.gif'; 
  
  // ★ ▼ ここに一匹だけ混ぜる別の生き物のURLを貼り付け ▼ ★
  const otherImageUrl = 'https://www.ishinazaka.co.jp/common/files/uploads/2026/05/1000006878.gif'; 

  const catCount = 10; 
  const otherIndex = -1; // 別画像の混入は無効化（URL未設定のため）

  for (let i = 0; i < catCount; i++) {
    setTimeout(() => {
      const cat = document.createElement('div');
      cat.className = 'horde-cat';
      
      // 画像の振り分け
      if (i === otherIndex) {
        cat.style.backgroundImage = `url('${otherImageUrl}')`;
      } else {
        cat.style.backgroundImage = `url('${catUrl}')`;
      }

      const size = Math.floor(Math.random() * 80) + 40;
      cat.style.width = `${size}px`;
      cat.style.height = `${size}px`;

      const duration = (Math.random() * 1.8) + 1.2;
      let startX, startY, endX, endY, flip;

      if (direction === 'bl-to-tr') {
        startX = Math.random() * 50 - 30; startY = Math.random() * 50 + 80; endX = Math.random() * 50 + 80; endY = Math.random() * 50 - 30; flip = 1; 
      } else if (direction === 'tr-to-bl') {
        startX = Math.random() * 50 + 80; startY = Math.random() * 50 - 30; endX = Math.random() * 50 - 30; endY = Math.random() * 50 + 80; flip = -1; 
      } else if (direction === 'bottom-to-top') {
        startX = Math.random() * 120 - 10; startY = 120; endX = startX + (Math.random() * 40 - 20); endY = -30; flip = Math.random() > 0.5 ? 1 : -1; 
      }

      container.appendChild(cat);

      cat.animate([
        { transform: `translate3d(${startX}vw, ${startY}vh, 0) scaleX(${flip})` },
        { transform: `translate3d(${endX}vw, ${endY}vh, 0) scaleX(${flip})` }
      ], {
        duration: duration * 1000,
        easing: 'linear',
        fill: 'forwards'
      });

      setTimeout(() => {
        cat.remove();
      }, duration * 1000);

    }, Math.random() * 800); // すぐに出現するように調整
  }
}

<h2 class="wp-block-heading has-white-color has-black-background-color has-text-color has-background has-link-color has-medium-font-size">家の周りに単粒度砕石を敷いて、防犯UP</h2>

<p class="has-medium-font-size"><strong>道路用砕石　A 5001　粒度の一覧表</strong></p>

<p class="has-medium-font-size">「ふるいに留まった骨材の質量百分率」とは、例えば骨材全部の質量(m<sub>all</sub>)が100ｇで、最も目の大きな40ｍｍのふるいに留まった骨材の質量(m<sub>40</sub>)が20ｇ、次に大きい20ｍｍのふるいに溜まった骨材の質量(m<sub>20</sub>)が10ｇの場合、以下のような計算になります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size">X40=m40/mall=20/100=20(％)</p>

<p class="has-medium-font-size">X20=（m40+m20）/mall=（20+10）/100=20+10=30(％)<br>ということになります。</p>
