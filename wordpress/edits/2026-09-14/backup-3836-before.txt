<script>
    let totalArea = 0; 
    let triangleCount = 0; 
    let formulaArray = []; 

    function getCircledNumber(num) {
        if (num >= 1 && num <= 20) {
            return String.fromCharCode(9311 + num);
        } else {
            return `(${num})`;
        }
    }

    function addTriangle() {
        const inputA = document.getElementById('sideA');
        const inputB = document.getElementById('sideB');
        const inputC = document.getElementById('sideC');
        
        const a = parseFloat(inputA.value);
        const b = parseFloat(inputB.value);
        const c = parseFloat(inputC.value);

        if (isNaN(a) || isNaN(b) || isNaN(c) || a <= 0 || b <= 0 || c <= 0) {
            alert("すべての辺に正の数値を入力してください。");
            return;
        }

        if (a + b <= c || a + c <= b || b + c <= a) {
            alert("三角形が成立しません。3辺の長さを確認してください。");
            return;
        }

        const s = (a + b + c) / 2;
        const area = Math.sqrt(s * (s - a) * (s - b) * (s - c));

        triangleCount++;
        const circleNum = getCircledNumber(triangleCount);
        formulaArray.push(`${circleNum} ${area.toFixed(2)}㎡`);
        document.getElementById('formulaDisplay').innerText = formulaArray.join(" ＋ ");

        totalArea += area;
        document.getElementById('totalAreaDisplay').innerText = totalArea.toFixed(2);
        
        // 再計算を促すため数値をリセット
        document.getElementById('baseVolumeDisplay').innerText = "0.00";
        document.getElementById('totalVolumeDisplay').innerText = "0.00";

        inputA.value = '';
        inputB.value = '';
        inputC.value = '';
        inputA.focus();
    }

    function calculateVolume() {
        if (totalArea === 0) {
            alert("先にSTEP1で面積を計算して追加してください。");
            return;
        }

        let thicknessCm = parseFloat(document.getElementById('thickness').value);
        if (isNaN(thicknessCm) || thicknessCm <= 0) {
            alert("正しい厚さを入力してください。");
            return;
        }
        
        // ★ 純粋な体積（面積 × 厚さ）
        const baseVolume = totalArea * (thicknessCm / 100);
        
        // ★ 転圧・食い込み割増（×1.26）
        const orderVolume = baseVolume * 1.26;
        
        // それぞれ画面に表示
        document.getElementById('baseVolumeDisplay').innerText = baseVolume.toFixed(2);
        document.getElementById('totalVolumeDisplay').innerText = orderVolume.toFixed(2);
    }

    function resetCalculator() {
        if(confirm("すべての履歴と計算結果をリセットしてもよろしいですか？")) {
            totalArea = 0;
            triangleCount = 0;
            formulaArray = []; 
            document.getElementById('formulaDisplay').innerText = "ここに計算式が表示されます";
            document.getElementById('totalAreaDisplay').innerText = "0.00";
            document.getElementById('baseVolumeDisplay').innerText = "0.00";
            document.getElementById('totalVolumeDisplay').innerText = "0.00";
            document.getElementById('sideA').value = '';
            document.getElementById('sideB').value = '';
            document.getElementById('sideC').value = '';
            document.getElementById('thickness').value = '5';
        }
    }
</script>

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size"><strong>なぜ「計算より多め」に必要なの?</strong></p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size">計算ツールを使っていただいた方の中には、「面積×厚さの計算結果よりも、実際の発注目安量の方が多いそ?」と気づかれた方もいる かもしれません。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size">実はここに、DIYで一番失敗しやすい落とし穴があります！</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size">砕石や砂利は、敷き詰めたあとに車が乗ったり、上から踏み固めたり（転圧といいます）すると、石同士の隙間がギュッと詰まったり、下の土に食い込んだりして、カサが大きく減ってしまいます。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size">そのため、ピッタリの量で注文してしまうと「あれ？敷いてみたら全然足りない…」という悲しい事態に。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size">この食い込み率は転圧を想定したものです。転圧をしない場合でも、長い時間をかけて、この係数くらいは下がります。</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"fontSize":"medium"} -->
<p class="has-medium-font-size">当社の計算ツールでは、そうした<strong>「転圧・食い込みによる目減り分」として、基準である【1.26倍】を自動で掛け算</strong>しています。この「発注目安量」を目安にご注文いただければ、途中で砕石が足りなくなる心配はありません！</p>
